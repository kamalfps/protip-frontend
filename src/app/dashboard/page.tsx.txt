"use client";

import React, { useEffect, useState } from 'react';
import { useRouter } from 'next/navigation';
import { fetchWithAuth } from '../../utils/api';

interface Donation {
  id: string;
  donorName: string;
  amount: number;
  message: string | null;
  createdAt: string;
}

interface WidgetSettings {
  alertImage: string;
  alertSound: string;
  alertDuration: number;
  textColor: string;
}

export default function DashboardPage() {
  const router = useRouter();
  const [username, setUsername] = useState('');
  const [donations, setDonations] = useState<Donation[]>([]);
  const [totalRevenue, setTotalRevenue] = useState(0);
  
  // Custom Widget State
  const [alertImage, setAlertImage] = useState('');
  const [alertSound, setAlertSound] = useState('');
  const [alertDuration, setAlertDuration] = useState(5000);
  const [textColor, setTextColor] = useState('#a855f7');
  
  const [loading, setLoading] = useState(true);
  const [saveStatus, setSaveStatus] = useState('');
  const [copied, setCopied] = useState(false);

  useEffect(() => {
    const token = localStorage.getItem('token');
    const storedUser = localStorage.getItem('username');
    
    if (!token || !storedUser) {
      router.push('/login');
      return;
    }
    
    setUsername(storedUser);

    async function loadDashboardData() {
      try {
        const res = await fetchWithAuth('/api/dashboard/stats');
        if (!res.ok) throw new Error('Unauthorized platform access');
        const data = await res.json();
        
        setDonations(data.donations || []);
        setTotalRevenue(data.totalRevenue || 0);
        
        if (data.settings) {
          setAlertImage(data.settings.alertImage || '');
          setAlertSound(data.settings.alertSound || '');
          setAlertDuration(data.settings.alertDuration || 5000);
          setTextColor(data.settings.textColor || '#a855f7');
        }
      } catch (err) {
        console.error(err);
      } finally {
        setLoading(false);
      }
    }

    loadDashboardData();
  }, [router]);

  const handleSaveSettings = async (e: React.FormEvent) => {
    e.preventDefault();
    setSaveStatus('Saving changes...');

    try {
      const res = await fetchWithAuth('/api/dashboard/settings', {
        method: 'POST',
        body: JSON.stringify({ alertImage, alertSound, alertDuration, textColor }),
      });

      if (res.ok) {
        setSaveStatus('Settings successfully saved!');
        setTimeout(() => setSaveStatus(''), 3000);
      } else {
        setSaveStatus('Error saving configurations.');
      }
    } catch (err) {
      setSaveStatus('Network writing failure.');
    }
  };

  const copyWidgetUrl = () => {
    const origin = typeof window !== 'undefined' ? window.location.origin : '';
    const url = `${origin}/widget/${username}`;
    navigator.clipboard.writeText(url);
    setCopied(true);
    setTimeout(() => setCopied(false), 2000);
  };

  const logout = () => {
    localStorage.removeItem('token');
    localStorage.removeItem('username');
    router.push('/login');
  };

  if (loading) {
    return (
      <div className="min-h-screen bg-slate-950 flex items-center justify-center text-white font-medium">
        Loading creator workspace metrics...
      </div>
    );
  }

  return (
    <div className="min-h-screen bg-slate-950 text-white flex flex-col md:flex-row">
      
      {/* Sidebar navigation */}
      <aside className="w-full md:w-64 bg-slate-900 border-b md:border-b-0 md:border-r border-slate-800 p-6 flex flex-col justify-between">
        <div>
          <div className="font-black text-2xl tracking-wider text-transparent bg-clip-text bg-gradient-to-r from-purple-400 to-pink-500 mb-8">
            PROTIP.LIVE
          </div>
          <div className="bg-slate-950/50 p-4 border border-slate-800 rounded-xl mb-6">
            <p className="text-xs text-slate-400 font-semibold uppercase tracking-wider">Creator</p>
            <p className="font-bold text-purple-300 text-lg">@{username}</p>
          </div>
        </div>
        <button onClick={logout} className="w-full text-left font-semibold text-sm text-slate-400 hover:text-red-400 py-2 transition-colors">
          Logout Session 🚪
        </button>
      </aside>

      {/* Main Panel Content Matrix */}
      <main className="flex-1 p-6 md:p-10 space-y-8 overflow-y-auto">
        
        {/* Row 1: Widget URL Core Engine Section */}
        <section className="bg-gradient-to-r from-purple-900/40 to-slate-900 border border-purple-500/20 rounded-2xl p-6">
          <h2 className="text-xl font-extrabold mb-2">Your Live OBS Browser Source Link</h2>
          <p className="text-sm text-purple-200/70 mb-4">Copy this secure endpoint and paste it as a Browser Source inside OBS Studio. Set width to 1920 and height to 1080.</p>
          <div className="flex flex-col sm:flex-row gap-3">
            <input 
              type="text" 
              readOnly 
              value={typeof window !== 'undefined' ? `${window.location.origin}/widget/${username}` : ''}
              className="flex-1 bg-slate-950/80 border border-slate-800 rounded-xl px-4 py-3 text-sm text-slate-300 font-mono outline-none"
            />
            <button 
              onClick={copyWidgetUrl}
              className="px-6 py-3 bg-purple-600 hover:bg-purple-700 font-bold rounded-xl text-sm transition-all shadow-md shrink-0"
            >
              {copied ? 'Copied Overlay! ✅' : 'Copy OBS URL'}
            </button>
          </div>
        </section>

        {/* Row 2: Performance Telemetry Analytics */}
        <section className="grid grid-cols-1 md:grid-cols-2 gap-6">
          <div className="bg-slate-900 border border-slate-800 rounded-2xl p-6">
            <p className="text-sm font-semibold text-slate-400 tracking-wider uppercase">Net Earnings Portfolio</p>
            <h3 className="text-5xl font-black text-emerald-400 tracking-tight mt-2">${totalRevenue.toFixed(2)}</h3>
          </div>
          <div className="bg-slate-900 border border-slate-800 rounded-2xl p-6">
            <p className="text-sm font-semibold text-slate-400 tracking-wider uppercase">Total Interactions</p>
            <h3 className="text-5xl font-black text-purple-400 tracking-tight mt-2">{donations.length}</h3>
          </div>
        </section>

        {/* Row 3: Live Real-Time Dashboard Customizer Matrix & Transaction Log */}
        <section className="grid grid-cols-1 lg:grid-cols-3 gap-8">
          
          {/* Customizer Panel */}
          <div className="lg:col-span-2 bg-slate-900 border border-slate-800 rounded-2xl p-6">
            <h3 className="text-lg font-bold mb-4">Alert Overlay Configuration Matrix</h3>
            
            <form onSubmit={handleSaveSettings} className="space-y-4">
              <div>
                <label className="block text-xs font-semibold text-slate-400 uppercase tracking-wider mb-1">Alert GIF Assets URL</label>
                <input 
                  type="text" 
                  value={alertImage} 
                  onChange={e => setAlertImage(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl px-4 py-3 text-sm text-slate-300 focus:outline-none focus:border-purple-500 transition-colors"
                />
              </div>

              <div>
                <label className="block text-xs font-semibold text-slate-400 uppercase tracking-wider mb-1">Alert Audio Sound URL (.mp3)</label>
                <input 
                  type="text" 
                  value={alertSound} 
                  onChange={e => setAlertSound(e.target.value)}
                  className="w-full bg-slate-950 border border-slate-800 rounded-xl px-4 py-3 text-sm text-slate-300 focus:outline-none focus:border-purple-500 transition-colors"
                />
              </div>

              <div className="grid grid-cols-2 gap-4">
                <div>
                  <label className="block text-xs font-semibold text-slate-400 uppercase tracking-wider mb-1">Active Duration (ms)</label>
                  <input 
                    type="number" 
                    value={alertDuration} 
                    onChange={e => setAlertDuration(Number(e.target.value))}
                    className="w-full bg-slate-950 border border-slate-800 rounded-xl px-4 py-3 text-sm text-slate-300 focus:outline-none focus:border-purple-500 transition-colors"
                  />
                </div>
                <div>
                  <label className="block text-xs font-semibold text-slate-400 uppercase tracking-wider mb-1">Typography Hex Color</label>
                  <div className="flex gap-2">
                    <input 
                      type="color" 
                      value={textColor} 
                      onChange={e => setTextColor(e.target.value)}
                      className="w-12 h-11 bg-transparent border-0 cursor-pointer"
                    />
                    <input 
                      type="text" 
                      value={textColor} 
                      onChange={e => setTextColor(e.target.value)}
                      className="flex-1 bg-slate-950 border border-slate-800 rounded-xl px-4 py-2 text-sm text-slate-300 uppercase focus:outline-none focus:border-purple-500"
                    />
                  </div>
                </div>
              </div>

              <button type="submit" className="w-full bg-purple-600 hover:bg-purple-700 text-white font-bold py-3 rounded-xl transition-all">
                Save Asset Parameters
              </button>
              {saveStatus && <p className="text-center text-sm font-semibold text-purple-400 mt-2">{saveStatus}</p>}
            </form>
          </div>

          {/* Ledger History Feed */}
          <div className="bg-slate-900 border border-slate-800 rounded-2xl p-6 flex flex-col">
            <h3 className="text-lg font-bold mb-4">Recent Inbound Donation Logs</h3>
            <div className="flex-1 overflow-y-auto space-y-3 max-h-[360px] pr-1">
              {donations.length === 0 ? (
                <p className="text-slate-500 text-sm text-center py-8">No inbound transfers recorded yet.</p>
              ) : (
                donations.map((donation) => (
                  <div key={donation.id} className="p-3 bg-slate-950/40 border border-slate-800 rounded-xl space-y-1">
                    <div className="flex justify-between items-center">
                      <span className="font-bold text-sm text-purple-300">{donation.donorName}</span>
                      <span className="font-black text-sm text-emerald-400">${donation.amount.toFixed(2)}</span>
                    </div>
                    {donation.message && <p className="text-xs text-slate-400 italic break-words">"{donation.message}"</p>}
                  </div>
                ))
              )}
            </div>
          </div>

        </section>

      </main>
    </div>
  );
}
